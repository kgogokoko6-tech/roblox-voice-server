const express = require("express"); const http = require("http"); const { Server } = require("socket.io"); const app = express(); const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*" } }); app.use(express.json()); const activeCodes = new Map(); const connectedUsers = new Map(); app.post("/generate-code", (req, res) => { const { userId, username } = req.body; if (!userId || !username) return res.status(400).json({ error: "بيانات غير مكتملة" }); const code = Math.random().toString(36).substring(2, 8).toUpperCase(); const expiresAt = Date.now() + 10 * 60 * 1000; activeCodes.set(code, { userId, username, expiresAt }); res.json({ success: true, code }); }); app.post("/update-positions", (req, res) => { const { players } = req.body; if (Array.isArray(players)) { io.emit("positions-update", players); } res.json({ status: "ok" }); }); app.get("/", (req, res) => { res.send(` VoiceBlox - Proximity Voice Chat
VoiceBlox
Connect your mic
Sign in with Roblox, or use the code shown in the game
OR
Codes come from the in-game panel and expire after 10 minutes. Signing in with Roblox stays linked for good.
متصل
VoiceBlox
شغال
القريبين منك
جاري البحث عن لاعبين...
`); }); io.on("connection", (socket) => { socket.on("verify-code", (code) => { const data = activeCodes.get(code); if (data && Date.now() < data.expiresAt) { connectedUsers.set(socket.id, data); activeCodes.delete(code); socket.emit("auth-success", data); } else { socket.emit("auth-failed", "الكود غير صحيح أو انتهت صلاحيته"); } }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => console.log("Voice Server is running on port " + PORT));
